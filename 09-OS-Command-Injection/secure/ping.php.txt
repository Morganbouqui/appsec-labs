<?php

$allowedHosts = [
    "8.8.8.8",
    "8.8.4.4",
    "1.1.1.1"
];

$host = $_GET['host'] ?? '';

if (!in_array($host, $allowedHosts, true)) {
    exit("Invalid host.");
}

?>

<!DOCTYPE html>
<html>
<head>
    <title>Secure Ping Result</title>
</head>

<body>

<h2>Ping Result</h2>

<pre>

<?php

if (PHP_OS_FAMILY === 'Windows') {

    system("ping -n 4 " . escapeshellarg($host));

} else {

    system("ping -c 4 " . escapeshellarg($host));

}

?>

</pre>

<br>

<a href="index.php">Back</a>

</body>
</html>
