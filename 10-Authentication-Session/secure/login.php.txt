<?php

/*
|--------------------------------------------------------------------------
| Secure Session Cookie Settings and very important 
|--------------------------------------------------------------------------
*/

session_set_cookie_params([
    'lifetime' => 0,
    'path' => '/',
    'secure' => true,   //this can be false if you running on http//
    'httponly' => true,
    'samesite' => 'Lax'
]);

session_start();

require 'db.php';

$error = "";

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';

    $stmt = $pdo->prepare("SELECT * FROM users WHERE username = ?");

    $stmt->execute([$username]);

    $user = $stmt->fetch(PDO::FETCH_ASSOC);

    if ($user && password_verify($password, $user['password'])) {

        /*
        |--------------------------------------------------------------------------
        | renegration session is set to true to stop session fixation
        |--------------------------------------------------------------------------
        */

        session_regenerate_id(true);

        $_SESSION['user'] = $user['username'];

        header("Location: dashboard.php");

        exit;

    } else {

        $error = "Invalid username or password.";

    }

}

?>

<!DOCTYPE html>

<html>

<head>

    <title>Secure Login</title>

</head>

<body>

<h2>Employee Login (Secure)</h2>

<?php if ($error): ?>

<p style="color:red;">

    <?= htmlspecialchars($error, ENT_QUOTES, 'UTF-8') ?>

</p>

<?php endif; ?>

<form method="POST">

    <label>

        Username

        <br>

        <input
            type="text"
            name="username"
            required
        >

    </label>

    <br><br>

    <label>

        Password

        <br>

        <input
            type="password"
            name="password"
            required
        >

    </label>

    <br><br>

    <button type="submit">

        Login

    </button>

</form>

</body>

</html>
