<?php

session_start();

/*This clear all session data */

$_SESSION = [];

/* Remove session cookie */

if (ini_get("session.use_cookies")) {

    $params = session_get_cookie_params();

    setcookie(

        session_name(),

        '',

        time() - 42000,

        $params["path"],

        $params["domain"],

        $params["secure"],

        $params["httponly"]

    );

}

/* Destroy session */

session_destroy();

header("Location: login.php");

exit;
